-- Polyfill task (Delta) - resto do script igual v13
if type(task) ~= "table" then
    task = {}
end
if type(task.wait) ~= "function" then
    task.wait = function(n)
        n = tonumber(n) or 0
        local rs = game:GetService("RunService")
        if n <= 0 then
            rs.Heartbeat:Wait()
            return
        end
        local t0 = tick()
        while tick() - t0 < n do
            rs.Heartbeat:Wait()
        end
    end
end
if type(task.delay) ~= "function" then
    task.delay = function(n, fn)
        task.spawn(function()
            task.wait(n)
            if type(fn) == "function" then fn() end
        end)
    end
end
if type(task.spawn) ~= "function" then
    task.spawn = function(fn)
        if type(fn) == "function" then
            coroutine.resume(coroutine.create(fn))
        end
    end
end
if type(task.defer) ~= "function" then
    task.defer = function(fn)
        task.spawn(fn)
    end
end

local function SalvarCoordenadas(tabela)
    local c = ""
    for nome, coords in pairs(tabela) do
        c = c .. string.format("%s|%.2f,%.2f,%.2f\n", nome, coords.x, coords.y, coords.z)
    end
    pcall(function() writefile("ball_pro_coords.txt", c) end)
end

local function CarregarCoordenadas()
    local ok, conteudo = pcall(function() return readfile("ball_pro_coords.txt") end)
    if not ok or not conteudo then return {} end
    local t = {}
    for linha in conteudo:gmatch("[^\n]+") do
        local nome, xy = linha:match("^(.+)|(.+)$")
        if nome and xy then
            local x, y, z = xy:match("([^,]+),([^,]+),([^,]+)")
            if x and y and z then
                t[nome] = {x = tonumber(x), y = tonumber(y), z = tonumber(z)}
            end
        end
    end
    return t
end
coordenadas = CarregarCoordenadas()

-- Troca rapida esquerdo/direito (v14)
local function NomeParAlternativo(nome)
    if not nome then return nil end
    local n = nome
    local pairList = {
        {"esquerdo", "direito"}, {"Esquerdo", "Direito"}, {"ESQUERDO", "DIREITO"},
        {"esquerda", "direita"}, {"Esquerda", "Direita"},
        {"left", "right"}, {"Left", "Right"}, {"LEFT", "RIGHT"},
        {"esq", "dir"}, {"Esq", "Dir"}, {"ESQ", "DIR"},
    }
    for _, p in ipairs(pairList) do
        if n:find(p[1], 1, true) then
            return n:gsub(p[1], p[2], 1)
        elseif n:find(p[2], 1, true) then
            return n:gsub(p[2], p[1], 1)
        end
    end
    return nil
end

local function TrocarAnguloCoordenada()
    local atual = estadoSistema.coordenadaSelecionada
    if not atual or atual == "" then
        Notify("Selecione uma coordenada primeiro", "warn")
        return
    end
    local alt = NomeParAlternativo(atual)
    if alt and coordenadas[alt] then
        local c = coordenadas[alt]
        estadoSistema.alvoAtual = Vector3.new(c.x, c.y, c.z)
        estadoSistema.coordenadaSelecionada = alt
        estadoSistema.guiaAtivo = false
        estadoSistema.cruzou = false
        if lblSelecao then lblSelecao.Text = "OK " .. alt end
        Notify("Trocou -> " .. alt, "ok")
        return
    end
    local prefix = atual:match("^(.*) ") or atual
    local lista = {}
    for nome, _ in pairs(coordenadas) do
        if nome:sub(1, #prefix) == prefix or (prefix ~= atual and nome:find(prefix, 1, true)) then
            table.insert(lista, nome)
        end
    end
    table.sort(lista)
    if #lista < 2 then
        Notify("Sem par esquerdo/direito para esta coord", "warn")
        return
    end
    local idx = 1
    for i, n in ipairs(lista) do
        if n == atual then idx = i break end
    end
    local nextName = lista[(idx % #lista) + 1]
    local c = coordenadas[nextName]
    estadoSistema.alvoAtual = Vector3.new(c.x, c.y, c.z)
    estadoSistema.coordenadaSelecionada = nextName
    estadoSistema.guiaAtivo = false
    estadoSistema.cruzou = false
    if lblSelecao then lblSelecao.Text = "OK " .. nextName end
    Notify("Trocou -> " .. nextName, "ok")
end

local cooldownToque = 0

local function CalcularDesvio(bola, dirAtual)
    if not desvioOn then return dirAtual end
    local melhorPlayer, melhorDist = nil, math.huge
    local bolaPos = bola.Position

    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= LocalPlayer and pl.Character then
            local hrp = pl.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local d = (hrp.Position - bolaPos).Magnitude
                if d <= raioDesvio and d < melhorDist then
                    local proPlayer = Vector3.new(hrp.Position.X - bolaPos.X, 0, hrp.Position.Z - bolaPos.Z)
                    if proPlayer.Magnitude > 0.1 then
                        local alinhamento = dirAtual:Dot(proPlayer.Unit)
                        if alinhamento > 0.3 then
                            melhorPlayer = hrp
                            melhorDist = d
                        end
                    end
                end
            end
        end
    end

    if not melhorPlayer then return dirAtual end

    local proPlayer = Vector3.new(melhorPlayer.Position.X - bolaPos.X, 0, melhorPlayer.Position.Z - bolaPos.Z).Unit
    local perp = Vector3.new(-dirAtual.Z, 0, dirAtual.X)
    local lado = perp:Dot(proPlayer) > 0 and -perp or perp
    local urgencia = math.clamp(1 - (melhorDist / raioDesvio), 0, 1)
    local dirDesvio = dirAtual:Lerp(lado, forcaDesvio * urgencia)
    return dirDesvio.Magnitude > 0.01 and dirDesvio.Unit or dirAtual
end

local function GuiarBola(bola, posGoal, dt)
    if not estadoSistema.ativo or not estadoSistema.alvoAtual then return end

    local vel = bola.AssemblyLinearVelocity
    local mag = Vector3.new(vel.X, 0, vel.Z).Magnitude

    -- Detectar chute + velocidade legitima (v14)
    if not estadoSistema.guiaAtivo and not estadoSistema.cruzou and mag >= VEL_CHUTE_MIN then
        estadoSistema.guiaAtivo = true
        estadoSistema.contandoParada = 0
        velChuteCapturada = math.clamp(mag * VEL_MULT_CHUTE, VEL_MINIMA, VEL_MAX_LEGIT)
        Notify(string.format("Chute detectado! vel: %.0f -> %.0f", mag, velChuteCapturada), "info")
    end
    if not estadoSistema.guiaAtivo then return end

    local distGoal = Vector3.new(posGoal.X - bola.Position.X, 0, posGoal.Z - bola.Position.Z)
    local dist = distGoal.Magnitude

    -- Depois de cruzar / dentro do raio → freia e auto-desliga
    if estadoSistema.cruzou or dist <= raioGoal then
        estadoSistema.cruzou = true
        local novoVel = bola.AssemblyLinearVelocity * freioPosGol
        if novoVel.Magnitude < 3 then
            bola.AssemblyLinearVelocity = Vector3.zero
            bola.AssemblyAngularVelocity = Vector3.zero
            estadoSistema.contandoParada = estadoSistema.contandoParada + 1
            if estadoSistema.contandoParada > TEMPO_PARA_DESLIGAR then
                estadoSistema.guiaAtivo = false
                estadoSistema.cruzou = false
                estadoSistema.contandoParada = 0
                estadoSistema.ativo = false
                Notify("Alvo alcançado — sistema desligado", "ok")
            end
            return
        end
        bola.AssemblyLinearVelocity = novoVel
        return
    end

    local velHoriz = Vector3.new(vel.X, 0, vel.Z)
    local apontandoProGoal = velHoriz.Magnitude > 1 and velHoriz.Unit:Dot(distGoal.Unit) > 0.4

    -- Zona perto do alvo: bola "morre" na frente
    if dist <= zonaGoal and apontandoProGoal then
        local novaMagZ = math.max(velHoriz.Magnitude - ATRITO_GOAL * dt, 0)
        if novaMagZ * dt * 60 < dist then
            bola.AssemblyLinearVelocity = Vector3.zero
            bola.AssemblyAngularVelocity = Vector3.zero
            estadoSistema.contandoParada = estadoSistema.contandoParada + 1
            if estadoSistema.contandoParada > TEMPO_PARA_DESLIGAR then
                estadoSistema.guiaAtivo = false
                estadoSistema.cruzou = false
                estadoSistema.contandoParada = 0
                estadoSistema.ativo = false
            end
            return
        end
        bola.AssemblyLinearVelocity = Vector3.new(velHoriz.Unit.X * novaMagZ, vel.Y, velHoriz.Unit.Z * novaMagZ)
        return
    end

    -- Proteção: nunca deixar passar do alvo
    local vaiPassar = dist < zonaGoal and velHoriz.Magnitude > 1 and velHoriz.Unit:Dot(distGoal.Unit) > 0.6

    -- Velocidade legitima (v14) + atrito natural
    local alvoVel = velChuteCapturada > 0 and velChuteCapturada or VEL_MINIMA
    if dist < 30 then
        alvoVel = alvoVel * (0.82 + 0.18 * math.clamp(dist / 30, 0, 1))
    end
    local novaMag = math.max(mag - ATRITO * dt, math.min(alvoVel, VEL_MAX_LEGIT))
    if velChuteCapturada > 0 then
        novaMag = math.min(novaMag, velChuteCapturada * 1.05)
    end

    -- Direção atual
    local dirAtual = velHoriz
    if dirAtual.Magnitude < 1 then
        dirAtual = distGoal.Unit
    else
        dirAtual = dirAtual.Unit
    end

    local dirFinal = dirAtual
    if not vaiPassar then
        local alinhamento = dirAtual:Dot(distGoal.Unit)
        local desalinhado = 1 - math.max(alinhamento, -1)

        -- FORCA_CURVA multiplica a correção (controle da UI)
        local fator = (CORRECAO_ALINHADO + desalinhado * CORRECAO_DESALINHADO) * FORCA_CURVA

        -- DIST_INICIO_CURVA: quanto mais perto dessa distância, mais forte a correção
        if dist < DIST_INICIO_CURVA then
            local t = math.clamp(1 - (dist / DIST_INICIO_CURVA), 0, 1)
            fator = fator * (0.7 + t * 0.6)
        end

        if dist < 25 then
            fator = fator * REDUCAO_PERTO
        end

        dirFinal = dirAtual:Lerp(distGoal.Unit, math.clamp(fator, 0, 0.45))
        if dirFinal.Magnitude > 0.01 then
            dirFinal = dirFinal.Unit
        else
            dirFinal = distGoal.Unit
        end
    end

    -- Desvio de players DEPOIS da correção (gera curva em S natural)
    dirFinal = CalcularDesvio(bola, dirFinal)

    local velY = bola.AssemblyLinearVelocity.Y
    bola.AssemblyLinearVelocity = Vector3.new(dirFinal.X * novaMag, velY, dirFinal.Z * novaMag)
    bola.AssemblyAngularVelocity = Vector3.zero

    if os.clock() - cooldownToque > 0.15 then
        cooldownToque = os.clock()
        TocarBola(bola)
    end
end

-- Detector de bola para o Follow (mais confiável - pega a mais próxima)
local function GetValidBallForFollow()
    if not RootPart then return nil end
    local closestBall = nil
    local shortestDist = math.huge

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local nameLower = obj.Name:lower()
            if nameLower == "tps" or nameLower == "ball" or nameLower:find("ball") or nameLower:find("soccer") then
                local size = obj.Size
                if size.X >= 0.8 and size.X <= 6.0 then
                    local dist = (obj.Position - RootPart.Position).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        closestBall = obj
                    end
                end
            end
        end
    end
    return closestBall
end

local function DoAutoFollow(ball)
    if not AutoFollowEnabled or IsManualInputActive() or not ball then return end
    if not RootPart or not RootPart.Parent or not Humanoid or not Humanoid.Parent then return end

    local targetPos = ball.Position
    local delta = targetPos - RootPart.Position
    local moveDirection = Vector3.new(delta.X, 0, delta.Z)
    local mag = moveDirection.Magnitude

    if mag > FollowStopDistance and mag < 150 then
        Humanoid:Move(moveDirection.Unit, false)
        RootPart.CFrame = CFrame.new(
            RootPart.Position,
            Vector3.new(targetPos.X, RootPart.Position.Y, targetPos.Z)
        )
    else
        Humanoid:Move(Vector3.zero, false)
    end
end

-- ================= SKILL MODE (do TH HUB) =================
local function DoSkillMode(dt)
    if not SkillModeEnabled then return end
    if not RootPart or not RootPart.Parent then return end

    local closest = GetValidBallForFollow()
    if not closest or not closest.Parent then return end

    local dist = (closest.Position - RootPart.Position).Magnitude
    if dist > 12 then return end

    skillTimer = skillTimer + (dt or 0.016) * SkillSpeed
    if skillTimer >= 1 then
        skillTimer = 0
        skillPhase = (skillPhase + 1) % 4
    end

    local t = skillTimer
    local offset
    if skillPhase == 0 then
        offset = RootPart.CFrame.LookVector * math.sin(t * math.pi) * 2
    elseif skillPhase == 1 then
        offset = RootPart.CFrame.RightVector * math.sin(t * math.pi) * 2
    elseif skillPhase == 2 then
        offset = -RootPart.CFrame.RightVector * math.sin(t * math.pi) * 2
    else
        offset = -RootPart.CFrame.LookVector * math.sin(t * math.pi) * 2
    end

    local targetPos = RootPart.Position + offset + Vector3.new(0, -1.5, 0)
    closest.CFrame = CFrame.new(closest.Position:Lerp(targetPos, 0.3), closest.Position)
    closest.AssemblyLinearVelocity = Vector3.zero
    closest.AssemblyAngularVelocity = Vector3.zero
end

local function DoReach()
    if not ReachEnabled then return end
    if not Character or not RootPart or not RootPart.Parent then return end

    -- Todas as partes relevantes (R6 + R15)
    local bodyParts = { RootPart }
    local names = {
        "Torso", "UpperTorso", "LowerTorso", "Head",
        "Right Arm", "Left Arm", "Right Leg", "Left Leg",
        "RightHand", "LeftHand", "RightFoot", "LeftFoot",
        "RightUpperArm", "LeftUpperArm", "RightUpperLeg", "LeftUpperLeg"
    }
    for _, name in ipairs(names) do
        local p = Character:FindFirstChild(name)
        if p then table.insert(bodyParts, p) end
    end

    local maxDistSq = ReachValue * ReachValue

    for ballPart, _ in pairs(ballsTable) do
        if ballPart and ballPart.Parent then
            -- Usa a distância do RootPart (centro do círculo)
            local dx = RootPart.Position.X - ballPart.Position.X
            local dy = RootPart.Position.Y - ballPart.Position.Y
            local dz = RootPart.Position.Z - ballPart.Position.Z

            if (dx*dx + dy*dy + dz*dz) <= maxDistSq then
                -- Bola DENTRO do círculo → toca com todas as partes
                for _, bodyPart in ipairs(bodyParts) do
                    pcall(function()
                        firetouchinterest(bodyPart, ballPart, 0)
                        firetouchinterest(bodyPart, ballPart, 1)
                    end)
                end
                -- Toque extra forte no RootPart
                pcall(function()
                    firetouchinterest(RootPart, ballPart, 0)
                    firetouchinterest(RootPart, ballPart, 1)
                    firetouchinterest(RootPart, ballPart, 0)

local function UpdatePrediction(bola)
    if not PredictionEnabled or StealthMode then
        ClearPrediction()
        return
    end
    if not bola or not bola.Parent then
        ClearPrediction()
        return
    end

    local vel = bola.AssemblyLinearVelocity
    local speed = vel.Magnitude
    -- Só mostra se a bola estiver em movimento relevante
    if speed < 8 then
        ClearPrediction()
        return
    end

    if not predFolder or not predFolder.Parent then
        predFolder = Instance.new("Folder")
        predFolder.Name = "DzPredictionBeam"
        predFolder.Parent = workspace
    end

    -- Simula trajetória
    local points = {}
    local pos = bola.Position
    local v = vel
    local groundY = bola.Position.Y - 2 -- aproximação; ajusta no primeiro hit

    for i = 1, PRED_STEPS do
        v = v + PRED_GRAVITY * PRED_DT
        -- atrito leve no ar
        v = v * (1 - 0.012 * PRED_DT * 60)
        pos = pos + v * PRED_DT

        -- Para se “bater no chão”
        if pos.Y <= (bola.Position.Y - 15) and v.Y < 0 then
            table.insert(points, Vector3.new(pos.X, bola.Position.Y - 1, pos.Z))
            break
        end
        table.insert(points, pos)
    end

    if #points < 2 then
        ClearPrediction()
        return
    end

    -- Reutiliza ou cria beams
    local needed = #points - 1
    while #predBeams > needed do
        local extra = table.remove(predBeams)
        pcall(function() extra:Destroy() end)
    end

    for i = 1, needed do
        local p0 = points[i]
        local p1 = points[i + 1]
        local beamData = predBeams[i]

        if not beamData or not beamData.part0 or not beamData.part0.Parent then
            local a0 = Instance.new("Part")
            a0.Name = "PredA0"
            a0.Anchored = true
            a0.CanCollide = false
            a0.CanQuery = false
            a0.CanTouch = false
            a0.Transparency = 1
            a0.Size = Vector3.new(0.2, 0.2, 0.2)
            a0.Parent = predFolder

            local a1 = Instance.new("Part")
            a1.Name = "PredA1"
            a1.Anchored = true
            a1.CanCollide = false
            a1.CanQuery = false
            a1.CanTouch = false
            a1.Transparency = 1
            a1.Size = Vector3.new(0.2, 0.2, 0.2)
            a1.Parent = predFolder

            local att0 = Instance.new("Attachment", a0)
            local att1 = Instance.new("Attachment", a1)

            local beam = Instance.new("Beam")
            beam.Attachment0 = att0
            beam.Attachment1 = att1
            beam.Width0 = 0.35
            beam.Width1 = 0.15
            beam.FaceCamera = true
            beam.LightEmission = 0.6
            beam.LightInfluence = 0
            beam.Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 60, 60)),
                ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 200, 50))
            })
            beam.Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 0.15),
                NumberSequenceKeypoint.new(1, 0.55)
            })
            beam.Parent = a0

            beamData = { part0 = a0, part1 = a1, beam = beam }
            predBeams[i] = beamData
        end

        beamData.part0.CFrame = CFrame.new(p0)
        beamData.part1.CFrame = CFrame.new(p1)
    end
end

local function IsSameTeam(plr)
    if not plr or plr == LocalPlayer then return false end
    -- Mesmo Team object
    if LocalPlayer.Team and plr.Team and LocalPlayer.Team == plr.Team then
        return true
    end
    -- Mesmo TeamColor
    if LocalPlayer.TeamColor and plr.TeamColor and LocalPlayer.TeamColor == plr.TeamColor then
        return true
    end
    -- Sem times definidos: trata todos os outros como possíveis aliados
    if not LocalPlayer.Team and not LocalPlayer.TeamColor then
        return true
    end
    return false
end

-- Nick parcial: "dvv" encontra "dvvi9p" se for o único match
-- Também reconhece o robô de teste "robo kkkkkk"
local function FindAllyByPartialNick(partial)
    if not partial or partial == "" then return nil, "vazio" end
    local q = partial:lower()
    local matches = {}

    -- Robô de teste
    if TestBotRoot and TestBotRoot.Parent then
        local botName = "robo kkkkkk"
        if botName:sub(1, #q) == q or botName:find(q, 1, true) then
            table.insert(matches, "TEST_BOT")
        end
    end

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and IsSameTeam(plr) then
            local name = plr.Name:lower()
            local display = (plr.DisplayName or ""):lower()
            if name:sub(1, #q) == q or name:find(q, 1, true) or display:sub(1, #q) == q or display:find(q, 1, true) then
                table.insert(matches, plr)
            end
        end
    end
    if #matches == 0 then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                local name = plr.Name:lower()
                local display = (plr.DisplayName or ""):lower()
                if name:sub(1, #q) == q or name:find(q, 1, true) or display:sub(1, #q) == q or display:find(q, 1, true) then
                    table.insert(matches, plr)
                end
            end
        end
    end
    if #matches == 1 then return matches[1], "ok" end
    if #matches == 0 then return nil, "nenhum" end
    return nil, "multiplos", matches
end

local function ScoreAllyFreedom(plr, ballPos)
    local root = GetPlayerRoot(plr)
    if not root then return -math.huge end
    local pos = root.Position
    local distBall = (pos - ballPos).Magnitude
    if distBall < 4 or distBall > 120 then return -math.huge end

    -- Distância do inimigo mais próximo (maior = mais livre)
    local nearestEnemy = math.huge
    for _, other in ipairs(Players:GetPlayers()) do
        if other ~= LocalPlayer and other ~= plr and not IsSameTeam(other) then
            local oroot = GetPlayerRoot(other)
            if oroot then
                local d = (oroot.Position - pos).Magnitude
                if d < nearestEnemy then nearestEnemy = d end
            end
        end
    end
    if nearestEnemy == math.huge then nearestEnemy = 50 end

    -- Preferência: longe de inimigos, distância média da bola
    local distScore = 1 - math.abs(distBall - 25) / 50
    return nearestEnemy * 2 + distScore * 15
end

local function GetBestAlly(ballPos)
    -- Se tem filtro de nick, tenta ele primeiro
    if PassNickFilter and PassNickFilter ~= "" then
        local plr, status = FindAllyByPartialNick(PassNickFilter)
        if plr then return plr end
        if status == "multiplos" then return nil end
        -- se nenhum match, cai no automático
    end

    local best, bestScore = nil, -math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and IsSameTeam(plr) then
            local score = ScoreAllyFreedom(plr, ballPos)
            if score > bestScore then
                bestScore = score
                best = plr
            end
        end
    end
    -- fallback sem team
    if not best then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                local score = ScoreAllyFreedom(plr, ballPos)
                if score > bestScore then
                    bestScore = score
                    best = plr
                end
            end
        end
    end
    return best
end

local function TentarIniciarPasse(bola)
    if PassGuiaAtivo or not bola then return end
    PassGuiaAtivo = true
    PassChegou = false
    PassContandoParada = 0
    PassToolTriggered = false
    local mag0 = Vector3.new(bola.AssemblyLinearVelocity.X, 0, bola.AssemblyLinearVelocity.Z).Magnitude
    velPasseCapturada = math.clamp(math.max(mag0, VEL_PASSE_MIN) * VEL_MULT_PASSE, 12, VEL_MAX_LEGIT * 0.85)
    if not (PassTargetPlayer and GetPlayerRoot(PassTargetPlayer)) then
        PassTargetPlayer = GetBestAlly(bola.Position)
    end
    if PassTargetPlayer then
        Notify("Passe -> " .. AllyName(PassTargetPlayer), "ok")
    else
        Notify("Nenhum aliado encontrado", "warn")
        PassGuiaAtivo = false
    end
end

local function FinalizarPasse(motivo)
    PassGuiaAtivo = false
    PassChegou = false
    PassContandoParada = 0
    -- Mantém PassTargetPlayer se tinha nick travado; só limpa se era automático
    if not PassNickFilter or PassNickFilter == "" then
        PassTargetPlayer = nil
    end
    Notify(motivo or "Passe entregue", "ok")
end

local function GuiarPasse(bola, dt)
    if not PassEnabled or not bola then return end
    local vel = bola.AssemblyLinearVelocity
    local mag = Vector3.new(vel.X, 0, vel.Z).Magnitude

    if not PassGuiaAtivo then
        if PassToolTriggered or mag >= VEL_PASSE_MIN then
            TentarIniciarPasse(bola)
        end
    end
    if not PassGuiaAtivo then return end

    local root = GetPlayerRoot(PassTargetPlayer)
    if not root then
        FinalizarPasse("Alvo sumiu")
        return
    end

    local posGoal = root.Position
    local distVec = Vector3.new(posGoal.X - bola.Position.X, 0, posGoal.Z - bola.Position.Z)
    local dist = distVec.Magnitude
    local velH = Vector3.new(vel.X, 0, vel.Z)

    -- ===== CHEGADA (igual lógica do chute) =====
    if PassChegou or dist <= PASS_RAIO_CHEGADA then
        PassChegou = true
        local novoVel = bola.AssemblyLinearVelocity * freioPosGol
        if novoVel.Magnitude < 3 then
            bola.AssemblyLinearVelocity = Vector3.zero
            bola.AssemblyAngularVelocity = Vector3.zero
            PassContandoParada = PassContandoParada + 1
            if PassContandoParada > 12 then
                FinalizarPasse("Passe chegou em " .. AllyName(PassTargetPlayer))
            end
            return
        end
        bola.AssemblyLinearVelocity = novoVel
        return
    end

    -- Zona de freio perto do aliado
    local apontando = velH.Magnitude > 1 and distVec.Magnitude > 0.1 and velH.Unit:Dot(distVec.Unit) > 0.35
    if dist <= PASS_ZONA_FREIO and apontando then
        local novaMagZ = math.max(velH.Magnitude - ATRITO_GOAL * dt, 0)
        if novaMagZ * dt * 60 < dist then
            bola.AssemblyLinearVelocity = Vector3.zero
            bola.AssemblyAngularVelocity = Vector3.zero
            PassContandoParada = PassContandoParada + 1
            if PassContandoParada > 12 then
                FinalizarPasse("Passe chegou em " .. AllyName(PassTargetPlayer))
            end
            return
        end
        bola.AssemblyLinearVelocity = Vector3.new(velH.Unit.X * novaMagZ, vel.Y, velH.Unit.Z * novaMagZ)
        return
    end

    -- ===== CURVA DO PASSE + velocidade legitima (v14) =====
    local alvoVel = velPasseCapturada > 0 and velPasseCapturada or (VEL_MINIMA * 0.85)
    if dist < 25 then
        alvoVel = alvoVel * (0.80 + 0.20 * math.clamp(dist / 25, 0, 1))
    end
    local novaMag = math.max(mag - ATRITO * dt, math.min(alvoVel, VEL_MAX_LEGIT * 0.85))
    if velPasseCapturada > 0 then
        novaMag = math.min(novaMag, velPasseCapturada * 1.05)
    end
    local dirAtual = velH.Magnitude > 1 and velH.Unit or distVec.Unit

    local alinhamento = dirAtual:Dot(distVec.Unit)
    local desalinhado = 1 - math.max(alinhamento, -1)
    local fator = (CORRECAO_ALINHADO + desalinhado * CORRECAO_DESALINHADO) * PASS_FORCA_CURVA

    if dist < PASS_DIST_INICIO then
        local t = math.clamp(1 - (dist / PASS_DIST_INICIO), 0, 1)
        fator = fator * (0.7 + t * 0.6)
    end
    if dist < 22 then
        fator = fator * REDUCAO_PERTO
    end

    local dirFinal = dirAtual:Lerp(distVec.Unit, math.clamp(fator, 0, 0.45))
    if dirFinal.Magnitude > 0.01 then
        dirFinal = dirFinal.Unit
    else
        dirFinal = distVec.Unit
    end

    bola.AssemblyLinearVelocity = Vector3.new(dirFinal.X * novaMag, vel.Y, dirFinal.Z * novaMag)
    bola.AssemblyAngularVelocity = Vector3.zero
end

local function MakeDraggable(gui)
    local dragging, dragInput, dragStart, startPos
    gui.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = gui.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    gui.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            gui.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

    if StealthMode then
        print("[dzxzxzx] " .. tostring(msg))
        return
    end
    if not ScreenGui or not ScreenGui.Parent then
        print("[dzxzxzx] " .. tostring(msg))
        return
    end

    kind = kind or "info"
    local bg = COR_PRETO
    local accent = COR_VERMELHO
    if kind == "ok" then
        accent = Color3.fromRGB(60, 220, 100)
    elseif kind == "warn" then
        accent = Color3.fromRGB(255, 180, 40)
    elseif kind == "err" then
        accent = Color3.fromRGB(255, 60, 60)
    end

    local toast = Instance.new("Frame")
    toast.Size = UDim2.new(1, 0, 0, 22)
    toast.BackgroundColor3 = bg
    toast.BackgroundTransparency = 0.85
    toast.BorderSizePixel = 0
    toast.ZIndex = 201
    toast.Parent = toastHolder
    Instance.new("UICorner", toast).CornerRadius = UDim.new(0, 6)

    local stroke = Instance.new("UIStroke", toast)
    stroke.Color = accent
    stroke.Thickness = 1
    stroke.Transparency = 0.35

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, -12, 1, 0)
    lbl.Position = UDim2.new(0, 6, 0, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = tostring(msg)
    lbl.TextColor3 = COR_TEXTO
    lbl.TextSize = 11
    lbl.Font = Enum.Font.GothamBold
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.TextTruncate = Enum.TextTruncate.AtEnd
    lbl.ZIndex = 202
    lbl.Parent = toast

    -- Entra de cima
    toast.Position = UDim2.new(0, 0, 0, -10)
    task.spawn(function()
        task.wait(2.4)
        if toast and toast.Parent then
            toast:Destroy()
        end
    end)

    print("[dzxzxzx] " .. tostring(msg))
end

function AtualizarLista()
    for _, child in ipairs(scrollList:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end
    local count = 0
    for nome, coords in pairs(coordenadas) do
        count += 1
        local container = Instance.new("Frame")
        container.Size = UDim2.new(1, -4, 0, 30)
        container.BackgroundTransparency = 1
        container.Parent = scrollList

        local item = Instance.new("TextButton")
        item.Size = UDim2.new(1, -34, 0, 30)
        item.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
        item.Text = "  " .. nome
        item.TextSize = 11
        item.Font = Enum.Font.Gotham
        item.TextColor3 = COR_TEXTO
        item.TextXAlignment = Enum.TextXAlignment.Left
        item.BorderSizePixel = 0
        item.Parent = container
        item.AutoButtonColor = false
        Instance.new("UICorner", item).CornerRadius = UDim.new(0, 5)

        item.MouseButton1Click:Connect(function()
            estadoSistema.alvoAtual = Vector3.new(coords.x, coords.y, coords.z)
            estadoSistema.coordenadaSelecionada = nome
            estadoSistema.guiaAtivo = false
            estadoSistema.cruzou = false
            lblSelecao.Text = "✓ " .. nome
            Notify("Alvo: " .. nome, "ok")
        end)

        local btnDel = Instance.new("TextButton")
        btnDel.Size = UDim2.new(0, 30, 0, 30)
        btnDel.Position = UDim2.new(1, -30, 0, 0)
        btnDel.BackgroundColor3 = COR_VERMELHO
        btnDel.Text = "✕"
        btnDel.TextSize = 13
        btnDel.Font = Enum.Font.GothamBold
        btnDel.TextColor3 = COR_TEXTO_ESCURO
        btnDel.BorderSizePixel = 0
        btnDel.Parent = container
        btnDel.AutoButtonColor = false
        Instance.new("UICorner", btnDel).CornerRadius = UDim.new(0, 5)

        btnDel.MouseButton1Click:Connect(function()
            coordenadas[nome] = nil
            SalvarCoordenadas(coordenadas)
            AtualizarLista()
            if estadoSistema.coordenadaSelecionada == nome then
                lblSelecao.Text = "Nenhuma coordenada selecionada"
                estadoSistema.alvoAtual = nil
                estadoSistema.coordenadaSelecionada = nil
            end

UserInputService.InputBegan:Connect(function(input, gp)
    -- Captura de nova tecla (keybind customizável)
    if waitingKeybind and input.UserInputType == Enum.UserInputType.Keyboard then
        local key = input.KeyCode
        if key ~= Enum.KeyCode.Unknown then
            if waitingKeybind == "follow" then
                FOLLOW_KEY = key
                if btnKeyFollow then
                    btnKeyFollow.Text = "Tecla Follow: " .. KeyName(FOLLOW_KEY)
                    btnKeyFollow.BackgroundColor3 = COR_PRETO
                    btnKeyFollow.TextColor3 = COR_TEXTO
                end
                Notify("Tecla Follow: " .. KeyName(FOLLOW_KEY), "ok")
            elseif waitingKeybind == "stealth" then
                STEALTH_KEY = key
                if btnKeyStealth then
                    btnKeyStealth.Text = "Tecla Furtivo: " .. KeyName(STEALTH_KEY)
                    btnKeyStealth.BackgroundColor3 = COR_PRETO
                    btnKeyStealth.TextColor3 = COR_TEXTO
                end
                if lblStealth then
                    lblStealth.Text = "MODO FURTIVO: " .. KeyName(STEALTH_KEY) .. " / Mobile"
                end
                Notify("Tecla Furtivo: " .. KeyName(STEALTH_KEY), "ok")
            end
            waitingKeybind = nil
        end
        return
    end

    if gp then return end
    if input.KeyCode == FOLLOW_KEY then
        if not StealthMode then
            ToggleFollow()
        end
    elseif input.KeyCode == STEALTH_KEY then
        ToggleModoFurtivo()
    end
end)

local function RefreshPassTargetLabel()
    PassNickFilter = (passNickBox.Text or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if PassNickFilter == "" then
        lblPassTarget.Text = "Alvo: automático (mais livre)"
        return
    end
    local plr, status, matches = FindAllyByPartialNick(PassNickFilter)
    if plr then
        lblPassTarget.Text = "Alvo: " .. AllyName(plr)
        lblPassTarget.TextColor3 = Color3.fromRGB(80, 255, 120)
    elseif status == "multiplos" then
        local names = {}
        for i, m in ipairs(matches or {}) do
            if i <= 3 then table.insert(names, AllyName(m)) end
        end
        lblPassTarget.Text = "Vários: " .. table.concat(names, ", ")
        lblPassTarget.TextColor3 = Color3.fromRGB(255, 180, 40)
    else
        lblPassTarget.Text = "Nenhum jogador com \"" .. PassNickFilter .. "\""
        lblPassTarget.TextColor3 = Color3.fromRGB(255, 80, 80)
    end
end

    RefreshPassTargetLabel()

local function TogglePass()
    PassNickFilter = (passNickBox.Text or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if not PassEnabled then
        if PassNickFilter ~= "" then
            local plr, status = FindAllyByPartialNick(PassNickFilter)
            if status == "nenhum" then
                Notify("Nick não encontrado", "warn")
                return
            end
            if status == "multiplos" then
                Notify("Vários nicks — seja mais específico", "warn")
                return
            end
            PassTargetPlayer = plr
            Notify("Passe travado em " .. AllyName(plr), "ok")
        else
            PassTargetPlayer = nil
            Notify("Passe no aliado mais livre", "ok")
        end
        PassEnabled = true
        PassGuiaAtivo = false
        PassToolTriggered = false
    else
        PassEnabled = false
        PassGuiaAtivo = false
        PassChegou = false
        PassContandoParada = 0
        PassTargetPlayer = nil
        PassToolTriggered = false
        Notify("Passe Inteligente OFF", "info")
    end
    AtualizarVisualPasse()
end
